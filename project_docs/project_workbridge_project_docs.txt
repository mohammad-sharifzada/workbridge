    image: postgres:17
    container_name: workbridge-db
    restart: unless-stopped
    environment:
      POSTGRES_USER: workbridge
      POSTGRES_PASSWORD: workbridge_dev_password
      POSTGRES_DB: workbridge
    ports:
      - "5432:5432"
    volumes:
      - workbridge_postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U workbridge -d workbridge"]
